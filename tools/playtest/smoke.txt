# Smoke test: boot, new game with the default challenge settings, save, power cycle, continue.
# Run from a blank save:
#   tools/playtest/playtest.py --rom pokehns.gba --session <dir> --fresh @tools/playtest/smoke.txt
# Any "FAIL" line (a screen that never came) makes the run exit 1. Look at the two shots.

mashuntil START CB2_MainMenu 3000       # intro and title screen
wait 90                                 # main menu fade-in eats an early A
press A                                 # NEW GAME
mashuntil A MainCB2 5000                # professor's speech up to the challenge menu (MainCB2 there)
wait 60                                 # its fade-in ignores input
press R; press R; press R; press R; press R; press R; press R   # last tab, CHALLENGES (MUTAGEN third, CHEATS fourth)
wait 30
press DOWN 4 8; press DOWN 4 8; press DOWN 4 8; press DOWN 4 8; press DOWN 4 8
press DOWN 4 8; press DOWN 4 8; press DOWN 4 8; press DOWN 4 8; press DOWN 4 8
press DOWN 4 8; press DOWN 4 8; press DOWN 4 8; press DOWN 4 8; press DOWN 4 8
press A                                 # SAVE, the last row
mashuntil A CB2_Overworld 4000          # rest of the speech; mashing A names the player AAAAAAA
wait 200
press START; wait 30
press DOWN 4 8; press DOWN 4 8          # BAG, <player>, SAVE
press A; wait 60
press A; wait 300                       # YES, then the write
press A 10 60; press B 10 60            # close the message
reboot                                  # power cycle: only the save RAM survives
mashuntil START CB2_MainMenu 2000
wait 90
shot continue                           # CONTINUE with the player's name, 0:00, 00 badges
press A
until CB2_Overworld 1200
wait 60
shot overworld                          # the same room and spot as when saved
